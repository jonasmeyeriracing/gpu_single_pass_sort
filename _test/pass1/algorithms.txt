# Algorithm registry for this shader directory (read by GpuSort at startup).
#
#   algorithm <name>
#   dispatch <file.hlsl> <entryPoint> <groupSize> [NAME=VALUE ...]
#
# Each 'dispatch' line is one ExecuteIndirect(DISPATCH {numSorts, 1, 1}) of that shader, compiled
# at runtime for cs_6_6 with -O3. The framework always adds -D GROUP_SIZE=<groupSize>; extra
# NAME=VALUE tokens become further defines (e.g. the size tier bounds MIN_COUNT / MAX_COUNT).
# One group handles one sort (SV_GroupID.x = sort index). The dispatches of an algorithm are
# issued back-to-back WITHOUT barriers, so each one must early-out for sorts outside its tier.
#
# Bindings (see common.hlsli): b0 root constants, t0 sort descriptors, t1 input, u0 output.

algorithm pass0_bitonic
dispatch pass0_bitonic.hlsl main 1024 MIN_COUNT=1 MAX_COUNT=8192

# pass1: size tiers. The large tier is the unchanged pass0 bitonic shader restricted by MIN_COUNT,
# so differences come from the tiering and the rank sort only. rank_sort.hlsl derives
# ELEMS_PER_THREAD = ceil(MAX_COUNT / GROUP_SIZE).

algorithm t2_rank512_bitonic
dispatch rank_sort.hlsl main 512 MIN_COUNT=1 MAX_COUNT=512
dispatch pass0_bitonic.hlsl main 1024 MIN_COUNT=513 MAX_COUNT=8192

algorithm t3_rank128_rank512_bitonic
dispatch rank_sort.hlsl main 128 MIN_COUNT=1 MAX_COUNT=128
dispatch rank_sort.hlsl main 512 MIN_COUNT=129 MAX_COUNT=512
dispatch pass0_bitonic.hlsl main 1024 MIN_COUNT=513 MAX_COUNT=8192

algorithm t3_rank64_rank512_bitonic
dispatch rank_sort.hlsl main 64 MIN_COUNT=1 MAX_COUNT=64
dispatch rank_sort.hlsl main 512 MIN_COUNT=65 MAX_COUNT=512
dispatch pass0_bitonic.hlsl main 1024 MIN_COUNT=513 MAX_COUNT=8192

# Mid tier as rank sort, 1 element per thread (513..1024), bitonic from 1025.
algorithm t4_rank128_rank512_rank1024_bitonic
dispatch rank_sort.hlsl main 128 MIN_COUNT=1 MAX_COUNT=128
dispatch rank_sort.hlsl main 512 MIN_COUNT=129 MAX_COUNT=512
dispatch rank_sort.hlsl main 1024 MIN_COUNT=513 MAX_COUNT=1024
dispatch pass0_bitonic.hlsl main 1024 MIN_COUNT=1025 MAX_COUNT=8192

# Mid tier as rank sort, 2 elements per thread (513..2048), bitonic from 2049.
algorithm t4_rank128_rank512_rank1024x2_bitonic
dispatch rank_sort.hlsl main 128 MIN_COUNT=1 MAX_COUNT=128
dispatch rank_sort.hlsl main 512 MIN_COUNT=129 MAX_COUNT=512
dispatch rank_sort.hlsl main 1024 MIN_COUNT=513 MAX_COUNT=2048
dispatch pass0_bitonic.hlsl main 1024 MIN_COUNT=2049 MAX_COUNT=8192
