# Algorithm registry for this shader directory (read by GpuSort at startup).
#
#   algorithm <name>
#   dispatch <file.hlsl> <entryPoint> <groupSize> [NAME=VALUE ...]
#   flush <full|full_legacy|full_ro|code|data|none>   (optional; overrides --flush-mode for this algorithm)
#
# Each 'dispatch' line is one ExecuteIndirect(DISPATCH {numSorts, 1, 1}) of that shader, compiled
# at runtime for cs_6_6 with -O3 (+ GROUP_SIZE, WAVE_SIZE, WAVE_SIZE_REQUIRED; see common.hlsli).
# One group handles one sort (SV_GroupID.x = sort index). The dispatches of an algorithm are issued
# back-to-back WITHOUT barriers, so each one early-outs for sorts outside its size range
# (MIN_COUNT..MAX_COUNT, inclusive) and the GPU may run them concurrently.
#
# pass7 set: low-end / integrated GPUs (AMD Ryzen iGPU, Intel UHD 770). See notes.md for the
# hypothesis behind every variant and what result would confirm or refute it. Short names:
#   R = rank sort, B<E> = register bitonic with E elements per thread, P = pass2 radix (1024
#   threads, 32 KB), X = pass7 radix (p7_radix.hlsli), @<n> = threads per group. Multi-dispatch
#   algorithms list the large tier first (it starts first, the small tiers fill in around it).
# p7_sort.hlsl sizes its groupshared array per dispatch: the next power of two >= MAX_COUNT.

# --- anchors -----------------------------------------------------------------------------------
# The portable reference (pass2 code): 1 dispatch @1024, 32 KB: R <= 512, B8 <= 2048, P above.
algorithm s1_rank512_bitreg2048_radix
dispatch single_pass.hlsl main 1024 RANK_MAX=512 BITREG_MAX=2048 LARGE_RADIX=1

# P for every size: the radix fixed cost (Intel: 117 us on mostly_empty).
algorithm s1_radix
dispatch single_pass.hlsl main 1024 RANK_MAX=0 LARGE_RADIX=1

# --- X for every size: its fixed cost and the effect of the group size --------------------------
algorithm x7_all_256
dispatch p7_sort.hlsl main 256 P7_LARGE=2

algorithm x7_all_1024
dispatch p7_sort.hlsl main 1024 P7_LARGE=2

# --- the reference's algorithms, one dispatch per tier with a group and LDS that fit the tier ---
# P 2049+ @1024 (32 KB) | B8 513-2048 @256 (8 KB) | R <= 512 @512 (2 KB)
algorithm m3_ref
dispatch radix_sort.hlsl main 1024 MIN_COUNT=2049
dispatch p7_sort.hlsl main 256 MIN_COUNT=513 MAX_COUNT=2048 P7_BIT_MAX=2048 P7_BIT_E=8
dispatch p7_sort.hlsl main 512 MAX_COUNT=512 P7_RANK_MAX=512

# The same with 32 KB of groupshared memory in every dispatch: group size alone vs LDS size.
algorithm m3_ref_32k
dispatch radix_sort.hlsl main 1024 MIN_COUNT=2049
dispatch p7_sort.hlsl main 256 MIN_COUNT=513 MAX_COUNT=2048 P7_BIT_MAX=2048 P7_BIT_E=8 P7_LDS_WORDS=8192
dispatch p7_sort.hlsl main 512 MAX_COUNT=512 P7_RANK_MAX=512 P7_LDS_WORDS=8192

# --- large tier: X in small groups ----------------------------------------------------------------
# X 2049+ @256 | B8 513-2048 @256 | R <= 512 @512
algorithm m3_x256
dispatch p7_sort.hlsl main 256 MIN_COUNT=2049 P7_LARGE=2
dispatch p7_sort.hlsl main 256 MIN_COUNT=513 MAX_COUNT=2048 P7_BIT_MAX=2048 P7_BIT_E=8
dispatch p7_sort.hlsl main 512 MAX_COUNT=512 P7_RANK_MAX=512

# X 2049+ @512 | B8 513-2048 @256 | R <= 512 @512
algorithm m3_x512

dispatch p7_sort.hlsl main 256 MIN_COUNT=513 MAX_COUNT=2048 P7_BIT_MAX=2048 P7_BIT_E=8
dispatch p7_sort.hlsl main 512 MAX_COUNT=512 P7_RANK_MAX=512

# Radix switch at 4096 with a 512-thread bitonic: X 4097+ @256 | B8 513-4096 @512 | R <= 512 @512

dispatch p7_sort.hlsl main 512 MAX_COUNT=512 P7_RANK_MAX=512

# --- mid tier (513-2048) --------------------------------------------------------------------------
# B4 @512 instead of B8 @256 (more threads, half the registers): X 2049+ @256 | B4 513-2048 @512 | R @512

# m4_b128's small tiers with the reference's large tier P @1024 (fallback if X loses)

# --- single dispatch with smaller groups (32 KB groupshared, one dispatch as the reference) -------
