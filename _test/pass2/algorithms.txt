# Algorithm registry for this shader directory (read by GpuSort at startup).
#
#   algorithm <name>
#   dispatch <file.hlsl> <entryPoint> <groupSize> [NAME=VALUE ...]
#
# Each 'dispatch' line is one ExecuteIndirect(DISPATCH {numSorts, 1, 1}) of that shader, compiled
# at runtime for cs_6_6 with -O3. The framework always adds -D GROUP_SIZE=<groupSize> and
# -D WAVE_SIZE=<device WaveLaneCountMin> (plus WAVE_SIZE_REQUIRED=1 if the device reports a range of
# wave sizes; see common.hlsli); extra NAME=VALUE tokens become further defines (e.g. the size tier
# bounds MIN_COUNT / MAX_COUNT).
# One group handles one sort (SV_GroupID.x = sort index). The dispatches of an algorithm are
# issued back-to-back WITHOUT barriers, so each one must early-out for sorts outside its tier.
#
# Bindings (see common.hlsli): b0 root constants, t0 sort descriptors, t1 input, u0 output.

# --- references (unchanged shaders from pass0 / pass1) -------------------------------------------

algorithm pass0_bitonic
dispatch pass0_bitonic.hlsl main 1024 MIN_COUNT=1 MAX_COUNT=8192

algorithm t2_rank512_bitonic
dispatch rank_sort.hlsl main 512 MIN_COUNT=1 MAX_COUNT=512
dispatch pass0_bitonic.hlsl main 1024 MIN_COUNT=513 MAX_COUNT=8192

# --- pass2: 2 dispatches, rank 512 + a new large tier ---------------------------------------------

# Register/wave bitonic, 8 elements per thread, up to 1024 threads.
algorithm t2_rank512_bitreg
dispatch rank_sort.hlsl main 512 MIN_COUNT=1 MAX_COUNT=512
dispatch bitonic_reg.hlsl main 1024 MIN_COUNT=513 MAX_COUNT=8192

# Register/wave bitonic, 16 elements per thread, up to 512 threads.
algorithm t2_rank512_bitreg16
dispatch rank_sort.hlsl main 512 MIN_COUNT=1 MAX_COUNT=512
dispatch bitonic_reg.hlsl main 512 MIN_COUNT=513 MAX_COUNT=8192 BR_ELEMS=16

# Register/wave bitonic, 32 elements per thread, up to 256 threads.
algorithm t2_rank512_bitreg32
dispatch rank_sort.hlsl main 512 MIN_COUNT=1 MAX_COUNT=512
dispatch bitonic_reg.hlsl main 256 MIN_COUNT=513 MAX_COUNT=8192 BR_ELEMS=32

# LDS radix sort (16-bit key, 4 x 4-bit passes), 1024 threads, up to 8 keys per thread.
algorithm t2_rank512_radix
dispatch rank_sort.hlsl main 512 MIN_COUNT=1 MAX_COUNT=512
dispatch radix_sort.hlsl main 1024 MIN_COUNT=513 MAX_COUNT=8192

# Same, but the wave scans use WavePrefixSum / WaveActiveSum instead of WaveReadLaneAt shuffles.
algorithm t2_rank512_radix_waveops
dispatch rank_sort.hlsl main 512 MIN_COUNT=1 MAX_COUNT=512
dispatch radix_sort.hlsl main 1024 MIN_COUNT=513 MAX_COUNT=8192 RS_WAVE_INTRINSICS=1

# --- pass2: 1 dispatch, 1024-thread group, group-uniform branch on count ---------------------------

# rank sort (1 element/thread) for count <= RANK_MAX, large-tier algorithm above.
algorithm s1_rank512_bitreg
dispatch single_pass.hlsl main 1024 RANK_MAX=512 LARGE_RADIX=0

algorithm s1_rank1024_bitreg
dispatch single_pass.hlsl main 1024 RANK_MAX=1024 LARGE_RADIX=0

algorithm s1_rank512_radix
dispatch single_pass.hlsl main 1024 RANK_MAX=512 LARGE_RADIX=1

algorithm s1_rank1024_radix
dispatch single_pass.hlsl main 1024 RANK_MAX=1024 LARGE_RADIX=1

# 3-way: rank sort <= 512, register bitonic up to BITREG_MAX, radix above.
algorithm s1_rank512_bitreg2048_radix
dispatch single_pass.hlsl main 1024 RANK_MAX=512 BITREG_MAX=2048 LARGE_RADIX=1

algorithm s1_rank512_bitreg4096_radix
dispatch single_pass.hlsl main 1024 RANK_MAX=512 BITREG_MAX=4096 LARGE_RADIX=1

# The large-tier algorithm for every size.
algorithm s1_bitreg
dispatch single_pass.hlsl main 1024 RANK_MAX=0 LARGE_RADIX=0

algorithm s1_radix
dispatch single_pass.hlsl main 1024 RANK_MAX=0 LARGE_RADIX=1
