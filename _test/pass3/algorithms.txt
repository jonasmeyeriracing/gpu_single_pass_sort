# Algorithm registry for this shader directory (read by GpuSort at startup).
#
#   algorithm <name>
#   dispatch <file.hlsl> <entryPoint> <groupSize> [NAME=VALUE ...]
#
# Each 'dispatch' line is one ExecuteIndirect(DISPATCH {numSorts, 1, 1}) of that shader, compiled
# at runtime for cs_6_6 with -O3. The framework always adds -D GROUP_SIZE=<groupSize> and
# -D WAVE_SIZE=<device WaveLaneCountMin> (plus WAVE_SIZE_REQUIRED=1 if the device reports a range of
# wave sizes; see common.hlsli); extra NAME=VALUE tokens become further defines (e.g. the size tier
# bounds MIN_COUNT / MAX_COUNT).
# One group handles one sort (SV_GroupID.x = sort index). The dispatches of an algorithm are
# issued back-to-back WITHOUT barriers, so each one must early-out for sorts outside its tier.
#
# Bindings (see common.hlsli): b0 root constants, t0 sort descriptors, t1 input, u0 output.
#
# pass3 full-run set (the diagnostic smoke-run set is _test/pass3/algorithms_diag.txt).

# --- references (pass2 shaders, unchanged) ---

algorithm s1_rank512_bitreg4096_radix
dispatch single_pass.hlsl main 1024 RANK_MAX=512 BITREG_MAX=4096 LARGE_RADIX=1

algorithm s1_rank512_bitreg2048_radix
dispatch single_pass.hlsl main 1024 RANK_MAX=512 BITREG_MAX=2048 LARGE_RADIX=1

# pass2 radix for every size (fixed-cost reference)
algorithm s1_radix
dispatch single_pass.hlsl main 1024 RANK_MAX=0 LARGE_RADIX=1

# --- pass3 ---

# radix_sort2 (min 8 keys per thread) for every size
algorithm s3_radix2_k8
dispatch single_pass3.hlsl main 1024 RANK_MAX=0 RADIX=2 RS2_MIN_KPT=8

# register bitonic with 4 elements per thread up to 4096, radix_sort2 above
algorithm s3_bitE4_rx2
dispatch single_pass3.hlsl main 1024 RANK_MAX=0 MID_E=4 MID_MAX=4096 RADIX=2 RS2_MIN_KPT=8

# rank <= 512, register bitonic (E = 8) <= 2048 / 4096, radix_sort2 above

algorithm s3_rank512_bitreg4096_radix2
dispatch single_pass3.hlsl main 1024 RANK_MAX=512 BITREG_MAX=4096 RADIX=2 RS2_MIN_KPT=8
